<script>

(function(){

  const API = "";

  const POLL_MS = 3 * 60 * 1000;

  let map, layer;

  const $ = id => document.getElementById(id);

  const statusEl = $("status");

  const btn = $("btnRefresh");


  function setStatus(msg, busy){

    statusEl.className = "status" + (busy ? " busy" : "");

    statusEl.innerHTML = busy ? msg : '<span class="live-dot"></span>' + msg;

  }

  function esc(s){

    return String(s == null ? "" : s)

      .replace(/&/g,"&amp;").replace(/</g,"&lt;").replace(/>/g,"&gt;").replace(/"/g,"&quot;");

  }

  function fmtAgo(iso, clock){

    if(!iso) return "—";

    const t = Date.parse(iso), n = clock ? Date.parse(clock) : Date.now();

    if(isNaN(t)) return iso;

    const sec = Math.max(0, Math.round((n - t) / 1000));

    if(sec < 60) return sec + "s ago";

    if(sec < 3600) return Math.round(sec/60) + "m ago";

    if(sec < 86400) return Math.round(sec/3600) + "h ago";

    return Math.round(sec/86400) + "d ago";

  }

  function fmtCoords(lat, lon){

    if(lat == null || lon == null) return "—";

    return Number(lat).toFixed(3) + ", " + Number(lon).toFixed(3);

  }

  function magClass(level, mag){

    if(level) return level;

    if(mag >= 7) return "escalate";

    if(mag >= 6) return "alert";

    if(mag >= 5.5) return "watch";

    return "list";

  }


  function initMap(){

    if(map) return;

    map = L.map("map", {scrollWheelZoom:false, worldCopyJump:true}).setView([20, 10], 2);

    L.tileLayer("https://{s}.basemaps.cartocdn.com/dark_all/{z}/{x}/{y}{r}.png", {

      attribution: "&copy; OSM &copy; CARTO", maxZoom: 18

    }).addTo(map);

    layer = L.layerGroup().addTo(map);

  }

  function renderMap(points){

    initMap();

    layer.clearLayers();

    const bounds = [];

    (points || []).forEach(p => {

      if(p.lat == null || p.lon == null) return;

      const mag = p.mag || 4.5;

      const r = Math.max(4, Math.min(16, (mag - 4) * 4));

      const color = mag >= 7 ? "#e85d6a" : mag >= 6 ? "#e8a54b" : mag >= 5.5 ? "#6b9eff" : "#8b9aab";

      const m = L.circleMarker([p.lat, p.lon], {

        radius:r, color:color, weight:1.5, fillColor:color, fillOpacity:0.55

      });

      m.bindPopup("<strong>M" + Number(mag).toFixed(1) + "</strong> · " + esc(p.place||"") + "<br/>" + esc(p.time||""));

      m.addTo(layer);

      bounds.push([p.lat, p.lon]);

    });

    if(bounds.length > 2){

      try{ map.fitBounds(bounds, {padding:[24,24], maxZoom:5}); }catch(e){}

    }

    setTimeout(() => map.invalidateSize(), 80);

  }


  function renderLookout(items){

    const box = $("lookout");

    if(!items || !items.length){

      box.innerHTML = '<div class="empty">No high-severity items in the current window.</div>';

      return;

    }

    box.innerHTML = items.map(it => {

      const lv = it.level || "watch";

      const link = it.url ? ' · <a href="'+esc(it.url)+'" target="_blank" rel="noopener">source</a>' : "";

      return '<div class="look-item '+esc(lv)+'"><div class="t">'+esc(it.title)+'</div><div class="m">'+

        esc(it.kind||"")+" · "+esc(it.meta||"")+link+"</div></div>";

    }).join("");

  }


  function renderQuakes(list, elId, clock){

    const box = $(elId);

    if(!list || !list.length){

      box.innerHTML = '<div class="empty">No M4.5+ events in this region (24h USGS).</div>';

      return;

    }

    box.innerHTML = list.slice(0, 25).map(q => {

      const mag = q.magnitude;

      const lv = magClass(q.level, mag);

      const maps = (q.latitude != null && q.longitude != null)

        ? "https://www.google.com/maps?q="+q.latitude+","+q.longitude : null;

      return '<div class="qrow"><div class="mag '+lv+'">'+(mag!=null?Number(mag).toFixed(1):"—")+'</div><div>'+

        '<div class="qtitle">'+esc(q.place)+'</div>'+

        '<div class="qmeta">'+esc(fmtAgo(q.time, clock))+" · "+esc(q.source||"USGS")+" · "+esc(q.status||"")+'</div>'+

        '<div class="details"><dl>'+

        "<dt>Time</dt><dd>"+esc(q.time||"—")+"</dd>"+

        "<dt>Coordinates</dt><dd>"+esc(fmtCoords(q.latitude, q.longitude))+"</dd>"+

        "<dt>Depth</dt><dd>"+(q.depth_km!=null?esc(q.depth_km)+" km":"—")+"</dd>"+

        "<dt>Status</dt><dd>"+esc(q.status||"—")+"</dd>"+

        "<dt>Source</dt><dd>"+esc(q.source||"USGS")+"</dd>"+

        "<dt>Maps</dt><dd>"+(maps?'<a href="'+maps+'" target="_blank" rel="noopener">Google Maps</a>':"—")+"</dd>"+

        "<dt>Detail</dt><dd>"+(q.url?'<a href="'+esc(q.url)+'" target="_blank" rel="noopener">Agency page</a>':"—")+"</dd>"+

        "</dl></div></div></div>";

    }).join("");

    box.querySelectorAll(".qrow").forEach(row => {

      row.addEventListener("click", () => row.classList.toggle("open"));

    });

  }


  function renderNotices(list, elId, emptyMsg){

    const box = $(elId);

    if(!list || !list.length){

      box.innerHTML = '<div class="empty">'+esc(emptyMsg)+"</div>";

      return;

    }

    box.innerHTML = list.slice(0, 12).map(it => {

      const lv = it.level || "watch";

      const title = it.headline || it.title || it.event || it.name || "Notice";

      const meta = [it.area, it.classification, it.source, it.time || it.onset || it.advisory].filter(Boolean).join(" · ");

      const url = it.url ? ' · <a href="'+esc(it.url)+'" target="_blank" rel="noopener">link</a>' : "";

      return '<div class="notice '+esc(lv)+'"><div class="n-t">'+esc(title)+'</div><div class="n-m">'+esc(meta)+url+"</div></div>";

    }).join("");

  }


  function renderSpace(sp){

    const box = $("space");

    if(!sp || (!sp.xray_class && sp.kp == null)){

      box.innerHTML = '<div class="empty">No SWPC reading yet.</div>';

      return;

    }

    box.innerHTML =

      "<div><span>X-ray class</span> · <strong>"+esc(sp.xray_class||"—")+"</strong></div>"+

      "<div><span>X-ray time</span> · "+esc(sp.xray_time||"—")+"</div>"+

      "<div><span>Kp index</span> · <strong>"+esc(sp.kp!=null?sp.kp:"—")+"</strong></div>"+

      "<div><span>Kp time</span> · "+esc(sp.kp_time||"—")+"</div>"+

      '<div style="margin-top:6px;font-size:.76rem;color:var(--muted)">NOAA SWPC GOES / planetary K-index</div>';

  }


  function renderSources(sources, ncs){

    const box = $("sources");

    const keys = Object.keys(sources || {});

    if(!keys.length){ box.innerHTML = "No source metadata."; return; }

    box.innerHTML = keys.map(k => {

      const s = sources[k] || {};

      const cnt = s.count != null ? " · " + s.count : "";

      return '<span class="src-pill '+(s.ok?"ok":"bad")+'">'+esc(k)+(s.ok?" ok":" fail")+cnt+"</span>";

    }).join(" ");

    const ncsEl = $("ncs-note");

    if(ncs && ncs.note){

      ncsEl.innerHTML = esc(ncs.note) + (ncs.url ? ' · <a href="'+esc(ncs.url)+'" target="_blank" rel="noopener">NCS</a>' : "");

    } else ncsEl.textContent = "";

  }


  function apply(data){

    const clock = data.evaluated_at;

    const sit = data.situation || {};

    $("c-escalate").textContent = sit.escalate ?? 0;

    $("c-alert").textContent = sit.alert ?? 0;

    $("c-watch").textContent = sit.watch ?? 0;

    $("c-listed").textContent = sit.listed ?? 0;

    $("c-space").textContent = sit.space ?? 0;

    renderLookout(data.lookout);

    renderMap(data.map_points);

    const q = data.quakes || {};

    renderQuakes(q.india, "q-india", clock);

    renderQuakes(q.us, "q-us", clock);

    renderQuakes(q.global, "q-global", clock);

    renderSpace(data.space);

    renderNotices(data.tornado, "tornado", "No active tornado warnings/watches.");

    renderNotices(data.tropical, "tropical", "No active NHC storms listed.");

    renderNotices((data.tsunami||[]).filter(t => t.level !== "info"), "tsunami", "No active PTWC/NTWC warning items.");

    renderNotices(data.severe, "severe", "No major severe-weather alerts in feed.");

    renderSources(data.sources, data.ncs);

    const iv = data.intervals || {};

    setStatus(

      "Updated "+(clock||"—")+" · UI poll 3 min · collect ~"+(iv.fast_minutes||15)+" min"+

      (data.elapsed_ms!=null?" · last collect "+data.elapsed_ms+" ms":""),

      false

    );

  }


  async function loadIntel(){

    setStatus("Gathering latest intelligence…", true);

    btn.disabled = true;

    try{

      const res = await fetch(API + "/api/intel", {cache:"no-store"});

      if(!res.ok) throw new Error("HTTP "+res.status);

      apply(await res.json());

    }catch(e){

      console.error(e);

      setStatus("Could not reach API. Start: ./run_server.sh (port 8080)", false);

    }finally{

      btn.disabled = false;

    }

  }


  btn.addEventListener("click", loadIntel);

  loadIntel();

  setInterval(loadIntel, POLL_MS);

})();

</script>

</body>
